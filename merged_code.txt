export async function listCarrers(req, res, next) {
  try {
    const page = parseInt(req.query.page) || 1;
    const limit = parseInt(req.query.limit) || 10;
    const name = req.query.name || '';

    console.log(`Listando carreras - Página: ${page}, Límite: ${limit}, Filtro: ${name}`);

    // Llamamos al servicio con paginación
    const result = await carrerService.fetchAllPaginated(page, limit, name);

    return res.status(200).json({
      success: true,
      message: 'XDXDDDD',
      data: {
        list: result.carrers,
        total: result.total,
        pages: result.totalPages,
        offset: result.offset,
        limit: limit,
        page: page
      },
      error: null
    });
  } catch (error) {
    console.error(error.stack);
    return res.status(500).json({
      success: false,
      message: 'Error al listar las carreras',
      data: null,
      error: error.message
    });
  }
}

/**
 * Guardar cambios múltiples (nuevos, editados, eliminados)
 */
export async function saveCarrers(req, res, next) {
  try {
    const { news = [], edits = [], deletes = [] } = req.body;

    console.log('Guardando cambios - Nuevos:', news.length, 'Editados:', edits.length, 'Eliminados:', deletes.length);

    // Procesar eliminaciones
    for (const id of deletes) {
      await carrerService.remove(id);
    }

    // Procesar ediciones
    for (const edit of edits) {
      await carrerService.update(edit.id, { name: edit.name });
    }

    // Procesar nuevas carreras
    const created = [];
    for (const newCarrer of news) {
      const createdCarrer = await carrerService.create({ name: newCarrer.name });
      // Guardar la relación entre el ID temporal y el real
      created.push({
        tmp: newCarrer.id,
        id: createdCarrer.id
      });
    }

    return res.status(200).json({
      success: true,
      message: 'Carreras guardadas con éxito',
      data: created,
      error: null
    });
  } catch (error) {
    console.error(error.stack);
    return res.status(500).json({
      success: false,
      message: 'Error al guardar las carreras',
      data: null,
      error: error.message
    });
  }
}

/**
 * Obtener una carrera por su ID (ej. /api/carrers/:id)
 */
export async function getCarrerById(req, res, next) {
  try {
    const { id } = req.params;
    const carrer = await carrerService.fetchById(id);

    // Validación si no existe el registro en la BD
    if (!carrer) {
      return res.status(404).json({
        success: false,
        message: 'Carrera no encontrada',
        data: null,
        error: `No existe la carrera con el id ${id}`
      });
    }

    return res.status(200).json({
      success: true,
      message: 'Carrera obtenida con éxito',
      data: carrer,
      error: null
    });
  } catch (error) {
    console.error(error.stack);
    return res.status(500).json({
      success: false,
      message: 'Error al buscar la carrera',
      data: null,
      error: error.message
    });
  }
}

// admin/models/application.js
import { DataTypes } from 'sequelize';
import sequelize from '../configs/database.js';

const Application = sequelize.define('Application', {
  id: {
    type: DataTypes.INTEGER,
    primaryKey: true,
    autoIncrement: true,
  },
  title: {
    type: DataTypes.STRING(40),
    allowNull: false,
  },
  description: {
    type: DataTypes.TEXT,
    allowNull: true,
  },
  benefits: {
    type: DataTypes.TEXT,
    allowNull: true,
  },
  functions: {
    type: DataTypes.TEXT,
    allowNull: true,
  },
  vacancies: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
  published: {
    type: DataTypes.DATE,
    allowNull: true,
  },
  email_notification: {
    type: DataTypes.BOOLEAN,
    defaultValue: false,
  },
  closed: {
    type: DataTypes.DATE,
    allowNull: true,
  },
  salarry: {
    type: DataTypes.DECIMAL(10, 2),
    allowNull: true,
  },
  is_draft: {
    type: DataTypes.BOOLEAN,
    defaultValue: true,
  },
  modality_id: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
  sementer_id: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
  district_id: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
  workday_id: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
  enterprise_id: {
    type: DataTypes.INTEGER,
    allowNull: true,
  },
}, {
  tableName: 'applications',
  timestamps: true,
  createdAt: 'created',
  updatedAt: false,
});

export default Application;

// admin/repositories/carrers_repository.js
import Carrer from '../models/carrer.js';
import { Op } from 'sequelize';

const showLogs = process.env.SHOW_DB_ERRORS === 'true' || process.env.NODE_ENV === 'development';

class CarrerRepository {
  /**
   * Obtiene todos los registros
   */
  async findAll() {
    try {
      return await Carrer.findAll();
    } catch (error) {
      if (showLogs) console.error('🔥 [CarrerRepo.findAll] Error:', error);
      throw error;
    }
  }

  /**
   * Obtiene registros con paginación
   */
  async findAllSimple(options = {}) {
    try {
      const { order = [['id', 'ASC']] } = options;
      return await Carrer.findAll({
        order
      });
    } catch (error) {
      if (showLogs) console.error('🔥 [CarrerRepo.findAllSimple] Error:', error);
      throw error;
    }
  }

  /**
   * Obtiene registros con paginación
   */
  async findAllPaginated(options = {}) {
    try {
      const { where = {}, offset = 0, limit = 10, order = [['id', 'DESC']] } = options;
      return await Carrer.findAll({
        where,
        offset,
        limit,
        order,
      });
    } catch (error) {
      if (showLogs) console.error('🔥 [CarrerRepo.findAllPaginated] Error:', error);
      throw error;
    }
  }

  /**
   * Cuenta registros con filtros
   */
  async count(where = {}) {
    try {
      return await Carrer.count({ where });
    } catch (error) {
      if (showLogs) console.error('🔥 [CarrerRepo.count] Error:', error);
      throw error;
    }
  }

  /**
   * Busca carrera por ID
   */
  async findById(id) {
    try {
      return await Carrer.findByPk(id);
    } catch (error) {
      if (showLogs) console.error(`🔥 [CarrerRepo.findById] Error (id: ${id}):`, error);
      throw error;
    }

export function hasRole(session, roleName) {
  if (!session || !Array.isArray(session.roles)) {
    return false;
  }

  return session.roles.some(role => 
    role?.name?.toLowerCase() === roleName?.toLowerCase()
  );
}

export function roleHasPermission(session, roleName, permissionName) {
  if (!session?.roles?.length || !roleName || !permissionName) {
    return false;
  }

  const normalizedRole = roleName.trim().toLowerCase();
  const normalizedPermission = permissionName.trim().toLowerCase();

  const role = session.roles.find(r =>
    r?.name?.trim().toLowerCase() === normalizedRole
  );

  if (!role?.permissions?.length) {
    return false;
  }

  return role.permissions.some(permission =>
    permission?.name?.trim().toLowerCase() === normalizedPermission
  );
}

// configs/middlewares.js

import { hasRole } from '../configs/helpers.js';

/**
 * Middleware para manejar errores 404
 * Si es una petición API (empieza con /api) devuelve JSON
 * Si es una petición web normal devuelve la vista 404
 */
export function notFoundHandler(req, res) {
  // Verificar si es una petición a la API
  if (req.path.startsWith('/api/')) {
    return res.status(404).json({
      success: false,
      message: 'Recurso no encontrado',
      data: null,
      error: 'Error 404'
    });
  }
  
  // Para peticiones no-API (y solo GET) renderizar vista 404
  if (req.method === 'GET') {
    return res.status(404).render('404', {
      title: 'Página no encontrada',
      message: 'La página que buscas no existe.'
    });
  }
  
  // Para otros métodos HTTP en rutas no-API
  return res.status(404).json({
    success: false,
    message: 'Recurso no encontrado',
    data: null,
    error: 'Error 404'
  });
}

/**
 * Middleware para manejo de errores generales
 * (Opcional - puedes expandirlo según necesites)
 */
export function errorHandler(err, req, res, next) {
  console.error('Error:', err.stack || err);
  
  // Determinar si es una petición API
  const isApiRequest = req.path.startsWith('/api/');
  
  const statusCode = err.statusCode || 500;
  
  if (isApiRequest) {
    return res.status(statusCode).json({
      success: false,
      message: err.message || 'Error interno del servidor',
      data: null,
      error: `Error ${statusCode}`
    });
  }
  
  // Para peticiones web
  res.status(statusCode).render('error', {
    title: 'Error',
    message: err.message || 'Ha ocurrido un error inesperado.',
    statusCode: statusCode
  });
}

export function requireAuth(req, res, next) {
  // Verificar si es una petición API
  const isApiRequest = req.path.startsWith('/api/');
  
  // Verificar si el usuario tiene sesión activa
  const isAuthenticated = req.session && req.session.user && req.session.user.id;
  if (isAuthenticated) {
    // Si está autenticado, continuar
    return next();
  }
  
  // Si NO está autenticado y es la ruta raíz '/'
  if (req.path === '/') {
    // Redireccionar a sign-in
    return res.redirect('/sign-in');
  }
  
  // Si NO está autenticado
  if (isApiRequest || req.method !== 'GET') {
    // Para APIs o métodos no-GET, devolver JSON
    return res.status(401).json({
      success: false,
      message: 'Acceso no autorizado',
      data: null,
      error: 'Error 401: No autenticado'
    });
  }
  
  // Para peticiones GET no-API, mostrar vista 401
  return res.status(401).render('401', {

export function redirectIfAuthenticated(req, res, next) {

    '/sign-in',

    '/login',
    '/register',
    '/reset-password'
  ].some(route => req.path === route);
  
  if (isAuthenticated && isAuthRoute) {
    // Redirigir al home si ya está logueado y quiere acceder a rutas de auth
    return res.redirect('/');

  // Si no está autenticado o no es una ruta de auth, continuar
  next();
}

// website/apis.js

export const sessionInfo = (req, res) => {
  // No hay sesión o no está autenticado
  if (!req.session || !req.session.user) {
    return res.status(401).json({
      success: false,
      message: 'No active session',
      data: null,
      error: 'SESSION_NOT_FOUND'
    });
  }

  // Hay sesión
  return res.json({
    success: true,
    message: 'Active session',
    data: {
      user: req.session.user
    }
  });
}

    return res.redirect("/" + userFound.role);
  });

export function logout(req, res) {
  req.session.destroy(err => {
    if (err) {
      console.error('Error al cerrar sesión:', err);
      return res.redirect('/');
    }

    // Limpia la cookie de sesión (opcional pero recomendado)
    res.clearCookie('connect.sid');

    // Redirige al login
    res.redirect('/login');
  });
}
